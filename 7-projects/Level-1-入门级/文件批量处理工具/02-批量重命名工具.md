## 第二部分 - 实现步骤

### 2.1 单个文件重命名

#### 概念说明

`Path.rename(target)` 将文件重命名为 `target`。`target` 必须是完整路径，不能只写文件名。

```
┌─────────────────────────────────────────────────────────────┐
│                  rename() 注意事项                           │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  ✅ 正确：file.rename(file.parent / "new_name.txt")         │
│     目标路径 = 父目录 / 新文件名                              │
│                                                             │
│  ❌ 错误：file.rename("new_name.txt")                        │
│     相对路径会相对于当前工作目录，不是文件所在目录             │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

#### 示例代码

```python
# rename_single.py
from pathlib import Path


def safe_rename(file: Path, new_name: str) -> Path:
    """安全重命名单个文件，返回新路径"""
    new_path = file.parent / new_name

    if new_path.exists():
        raise FileExistsError(f"目标文件已存在：{new_path}")

    file.rename(new_path)
    return new_path


# 使用示例
f = Path("/tmp/docs/report.txt")
new_f = safe_rename(f, "report_2024.txt")
print(f"重命名完成：{new_f}")
```

### 2.2 批量添加前缀

#### 概念说明

遍历目录中的所有文件，对每个文件名加上前缀，再调用 `rename()`。

#### 示例代码

```python
# add_prefix.py
from pathlib import Path


def add_prefix(
    directory: str,
    prefix: str,
    extensions: list[str] | None = None,
) -> list[str]:
    """批量添加文件名前缀

    Args:
        directory: 目标目录
        prefix: 要添加的前缀，如 "new_"
        extensions: 只处理指定扩展名，None 表示处理所有文件

    Returns:
        重命名记录列表，格式 "旧名 -> 新名"
    """
    dir_path = Path(directory)
    results = []

    for file in dir_path.iterdir():
        if not file.is_file():
            continue
        if extensions and file.suffix.lower() not in extensions:
            continue

        new_name = f"{prefix}{file.name}"
        new_path = file.parent / new_name
        file.rename(new_path)
        results.append(f"{file.name} -> {new_name}")

    return results


# 所有 .jpg 文件加上 "vacation_" 前缀
records = add_prefix("/tmp/photos", "vacation_", [".jpg"])
for r in records:
    print(r)
```

### 2.3 批量添加后缀

#### 概念说明

添加后缀时需要在**扩展名之前**插入，而不是直接追加到文件名末尾。

```
正确：photo.jpg  →  photo_edited.jpg
              ↑ 插在扩展名前
错误：photo.jpg  →  photo.jpg_edited
```

用 `Path.stem`（不含扩展名的文件名）+ 后缀 + `Path.suffix`（扩展名）拼接。

#### 示例代码

```python
# add_suffix.py
from pathlib import Path


def add_suffix(
    directory: str,
    suffix: str,
    extensions: list[str] | None = None,
) -> list[str]:
    """批量在扩展名前添加后缀"""
    dir_path = Path(directory)
    results = []

    for file in dir_path.iterdir():
        if not file.is_file():
            continue
        if extensions and file.suffix.lower() not in extensions:
            continue

        # stem = "photo", suffix = ".jpg"，新名 = "photo_edited.jpg"
        new_name = f"{file.stem}{suffix}{file.suffix}"
        new_path = file.parent / new_name
        file.rename(new_path)
        results.append(f"{file.name} -> {new_name}")

    return results


# 所有文件名加上 "_backup" 后缀
records = add_suffix("/tmp/docs", "_backup")
```

**关键代码说明：**

| 代码 | 含义 | 为什么这样写 |
|------|------|-------------|
| `file.stem` | 文件名不含扩展名部分 | `report.txt` 的 stem 是 `report` |
| `file.suffix` | 文件的扩展名含点号 | `report.txt` 的 suffix 是 `.txt` |
| `f"{file.stem}{suffix}{file.suffix}"` | 在 stem 和 suffix 之间插入后缀 | 保留原有扩展名，后缀加在它之前 |

### 2.4 批量添加时间戳

#### 概念说明

时间戳格式推荐 `YYYYMMDD_HHMMSS`，放在文件名末尾（扩展名之前），便于按时间排序。

#### 示例代码

```python
# add_timestamp.py
from pathlib import Path
from datetime import datetime


def add_timestamp(
    directory: str,
    extensions: list[str] | None = None,
) -> list[str]:
    """批量在文件名末尾添加当前时间戳"""
    # 所有文件使用同一个时间戳，保证批量操作的一致性
    ts = datetime.now().strftime("%Y%m%d_%H%M%S")
    dir_path = Path(directory)
    results = []

    for file in dir_path.iterdir():
        if not file.is_file():
            continue
        if extensions and file.suffix.lower() not in extensions:
            continue

        new_name = f"{file.stem}_{ts}{file.suffix}"
        new_path = file.parent / new_name
        file.rename(new_path)
        results.append(f"{file.name} -> {new_name}")

    return results


# 示例：note.md → note_20240115_143022.md
records = add_timestamp("/tmp/notes", [".md", ".txt"])
```

### 2.5 正则表达式替换文件名

#### 概念说明

`re.sub(pattern, replacement, string)` 将字符串中匹配正则的部分替换为 `replacement`。

#### 示例代码

```python
# regex_rename.py
import re
from pathlib import Path


def regex_rename(
    directory: str,
    pattern: str,
    replacement: str,
    extensions: list[str] | None = None,
) -> list[str]:
    """用正则表达式批量替换文件名中的内容

    Args:
        pattern: 正则模式，如 r"IMG_(\d+)"
        replacement: 替换内容，如 r"photo_\1"（\1 引用捕获组）
    """
    dir_path = Path(directory)
    results = []

    for file in dir_path.iterdir():
        if not file.is_file():
            continue
        if extensions and file.suffix.lower() not in extensions:
            continue

        # 只对文件名（不含扩展名）做替换
        new_stem = re.sub(pattern, replacement, file.stem)
        if new_stem == file.stem:
            continue  # 没有匹配，跳过

        new_name = f"{new_stem}{file.suffix}"
        new_path = file.parent / new_name
        file.rename(new_path)
        results.append(f"{file.name} -> {new_name}")

    return results


# IMG_001.jpg → vacation_001.jpg
records = regex_rename(
    "/tmp/photos",
    pattern=r"IMG_(\d+)",
    replacement=r"vacation_\1",
    extensions=[".jpg"],
)
```

**关键代码说明：**

| 代码 | 含义 | 为什么这样写 |
|------|------|-------------|
| `re.sub(pattern, replacement, file.stem)` | 只替换文件名，不动扩展名 | 防止把 `.jpg` 也被替换掉 |
| `r"IMG_(\d+)"` | 捕获 IMG_ 后面的数字 | `(\d+)` 是捕获组，可在 replacement 里用 `\1` 引用 |
| `if new_stem == file.stem: continue` | 没有匹配则跳过 | 避免对无关文件做无意义的 rename 操作 |

## 第三部分 - 安全重命名的注意事项

### 重命名前先预览

```python
# preview_rename.py
from pathlib import Path


def preview_rename(directory: str, prefix: str) -> list[tuple[str, str]]:
    """预览重命名结果（不实际执行）"""
    dir_path = Path(directory)
    preview = []
    for file in dir_path.iterdir():
        if file.is_file():
            new_name = f"{prefix}{file.name}"
            preview.append((file.name, new_name))
    return preview


# 先打印预览，确认无误再执行
for old, new in preview_rename("/tmp/photos", "vacation_"):
    print(f"  {old}  →  {new}")

confirm = input("\n确认重命名？(y/n): ")
if confirm.lower() == "y":
    from processor.operations import FileOperations
    FileOperations.add_prefix("/tmp/photos", "vacation_")
```

## 常见错误

### 1. 目标路径只写文件名

```python
# ❌ 错误：relative path，结果出现在当前工作目录
file.rename("new_report.txt")

# ✅ 正确：用 parent 保证在同一目录
file.rename(file.parent / "new_report.txt")
```

### 2. 没有检查目标文件是否已存在

```python
# ❌ 错误：若 new_report.txt 已存在，会静默覆盖
file.rename(file.parent / "new_report.txt")

# ✅ 正确：先检查
new_path = file.parent / "new_report.txt"
if new_path.exists():
    print(f"跳过，目标已存在：{new_path}")
else:
    file.rename(new_path)
```

### 3. 后缀加到扩展名后面

```python
# ❌ 错误：photo.jpg → photo.jpg_edited
new_name = file.name + "_edited"

# ✅ 正确：photo.jpg → photo_edited.jpg
new_name = f"{file.stem}_edited{file.suffix}"
```
