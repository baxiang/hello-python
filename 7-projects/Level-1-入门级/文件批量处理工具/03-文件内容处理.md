### 2.1 文件读写基础

#### 概念说明

Python 读写文件时必须指定编码，否则使用系统默认编码（macOS/Linux 是 UTF-8，Windows 可能是 GBK）。

```
┌─────────────────────────────────────────────────────────────┐
│                  open() 关键参数                             │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  mode（模式）：                                              │
│    "r"   读取（默认）                                        │
│    "w"   写入（覆盖）                                        │
│    "a"   追加                                               │
│    "rb"  二进制读取                                          │
│                                                             │
│  encoding（编码）：                                          │
│    "utf-8"   现代标准，推荐                                  │
│    "gbk"     中文 Windows 系统常用                           │
│    "utf-8-sig"  带 BOM 的 UTF-8（Excel 导出的 CSV 常用）      │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

#### 示例代码

```python
# file_rw.py
from pathlib import Path


def read_text(file_path: str, encoding: str = "utf-8") -> str:
    """读取文本文件内容"""
    return Path(file_path).read_text(encoding=encoding)


def write_text(file_path: str, content: str, encoding: str = "utf-8") -> None:
    """写入文本文件（覆盖）"""
    Path(file_path).write_text(content, encoding=encoding)


def append_text(file_path: str, content: str, encoding: str = "utf-8") -> None:
    """追加内容到文件"""
    with open(file_path, "a", encoding=encoding) as f:
        f.write(content)


# 示例
write_text("/tmp/hello.txt", "你好，世界！\n")
content = read_text("/tmp/hello.txt")
print(content)  # 你好，世界！
```

### 2.2 编码问题与检测

#### 概念说明

当文件编码未知时，用 `chardet` 库自动检测编码，避免 `UnicodeDecodeError`。

```
┌─────────────────────────────────────────────────────────────┐
│                  编码检测流程                                 │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  文件（二进制字节）                                          │
│       │                                                     │
│       ↓  chardet.detect()                                   │
│  {'encoding': 'GBK', 'confidence': 0.99}                    │
│       │                                                     │
│       ↓  用检测到的编码解码                                   │
│  正常的文本字符串                                            │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

安装依赖：`uv add chardet`

#### 示例代码

```python
# detect_encoding.py
import chardet
from pathlib import Path


def detect_encoding(file_path: str) -> str:
    """检测文件编码"""
    raw = Path(file_path).read_bytes()
    result = chardet.detect(raw)
    encoding = result.get("encoding") or "utf-8"
    confidence = result.get("confidence", 0)
    print(f"检测到编码：{encoding}（置信度 {confidence:.0%}）")
    return encoding


def read_auto(file_path: str) -> str:
    """自动检测编码并读取文件"""
    encoding = detect_encoding(file_path)
    return Path(file_path).read_text(encoding=encoding)


# 读取未知编码的文件
content = read_auto("/tmp/legacy_file.txt")
print(content[:100])
```

**关键代码说明：**

| 代码 | 含义 | 为什么这样写 |
|------|------|-------------|
| `Path(file_path).read_bytes()` | 以二进制模式读取文件 | 编码检测需要原始字节，不能提前解码 |
| `result.get("encoding") or "utf-8"` | 获取检测结果，检测失败时回退 UTF-8 | `chardet` 对极短文件可能返回 `None` |
| `confidence` | 检测置信度（0~1） | 低于 0.7 时检测结果不可信，可提示用户手动指定 |

### 2.3 批量文本搜索

#### 示例代码

```python
# search_content.py
from pathlib import Path


def search_in_file(file_path: str, keyword: str, encoding: str = "utf-8") -> list[tuple[int, str]]:
    """在单个文件中搜索关键词，返回 (行号, 行内容) 列表"""
    results = []
    lines = Path(file_path).read_text(encoding=encoding).splitlines()
    for line_no, line in enumerate(lines, start=1):
        if keyword in line:
            results.append((line_no, line.strip()))
    return results


def search_in_directory(
    directory: str,
    keyword: str,
    extensions: list[str] | None = None,
    encoding: str = "utf-8",
) -> dict[str, list[tuple[int, str]]]:
    """在目录下所有文件中搜索关键词"""
    dir_path = Path(directory)
    matches: dict[str, list[tuple[int, str]]] = {}

    for file in dir_path.rglob("*"):
        if not file.is_file():
            continue
        if extensions and file.suffix.lower() not in extensions:
            continue

        try:
            results = search_in_file(str(file), keyword, encoding)
            if results:
                matches[str(file)] = results
        except (UnicodeDecodeError, PermissionError):
            continue  # 跳过无法读取的文件

    return matches


# 在所有 Python 文件中搜索 "TODO"
results = search_in_directory("/project/src", "TODO", extensions=[".py"])
for file_path, hits in results.items():
    print(f"\n{file_path}")
    for line_no, line in hits:
        print(f"  L{line_no}: {line}")
```

### 2.4 批量文本替换

#### 示例代码

```python
# replace_content.py
from pathlib import Path


def replace_in_file(
    file_path: str,
    old_text: str,
    new_text: str,
    encoding: str = "utf-8",
) -> int:
    """替换单个文件中的文本，返回替换次数"""
    path = Path(file_path)
    content = path.read_text(encoding=encoding)

    if old_text not in content:
        return 0

    new_content = content.replace(old_text, new_text)
    count = content.count(old_text)
    path.write_text(new_content, encoding=encoding)
    return count


def replace_in_directory(
    directory: str,
    old_text: str,
    new_text: str,
    extensions: list[str] | None = None,
    encoding: str = "utf-8",
) -> dict[str, int]:
    """批量替换目录下所有文件中的文本

    Returns:
        {文件路径: 替换次数} 的字典，只包含有替换发生的文件
    """
    dir_path = Path(directory)
    results: dict[str, int] = {}

    for file in dir_path.rglob("*"):
        if not file.is_file():
            continue
        if extensions and file.suffix.lower() not in extensions:
            continue

        try:
            count = replace_in_file(str(file), old_text, new_text, encoding)
            if count > 0:
                results[str(file)] = count
        except (UnicodeDecodeError, PermissionError):
            continue

    return results


# 将所有 .txt 文件中的 "http://" 替换为 "https://"
replaced = replace_in_directory(
    "/project/docs",
    old_text="http://",
    new_text="https://",
    extensions=[".txt", ".md"],
)
for path, count in replaced.items():
    print(f"{path}: 替换了 {count} 处")
```

**关键代码说明：**

| 代码 | 含义 | 为什么这样写 |
|------|------|-------------|
| `if old_text not in content: return 0` | 没有匹配则直接返回 | 避免对无需修改的文件执行无意义的写操作 |
| `content.count(old_text)` | 统计替换次数（在 replace 之前） | replace 之后 old_text 已消失，无法再统计 |
| `except (UnicodeDecodeError, PermissionError)` | 跳过无法处理的文件 | 二进制文件或受保护文件不应导致整个批量任务失败 |

### 2.5 编码统一转换

#### 示例代码

```python
# convert_encoding.py
import chardet
from pathlib import Path


def convert_encoding(
    file_path: str,
    target_encoding: str = "utf-8",
) -> bool:
    """将文件转换为目标编码，返回是否执行了转换"""
    path = Path(file_path)
    raw = path.read_bytes()

    result = chardet.detect(raw)
    source_encoding = result.get("encoding") or "utf-8"

    # 已经是目标编码，跳过
    if source_encoding.lower().replace("-", "") == target_encoding.lower().replace("-", ""):
        return False

    content = raw.decode(source_encoding)
    path.write_text(content, encoding=target_encoding)
    print(f"{path.name}: {source_encoding} → {target_encoding}")
    return True


def convert_directory(directory: str, extensions: list[str] | None = None) -> int:
    """将目录下所有文件统一转为 UTF-8，返回转换文件数"""
    converted = 0
    for file in Path(directory).rglob("*"):
        if not file.is_file():
            continue
        if extensions and file.suffix.lower() not in extensions:
            continue
        try:
            if convert_encoding(str(file)):
                converted += 1
        except (UnicodeDecodeError, ValueError):
            print(f"跳过（转换失败）：{file.name}")
    return converted


# 将目录下所有文本文件统一转为 UTF-8
count = convert_directory("/tmp/legacy_docs", extensions=[".txt", ".csv"])
print(f"共转换 {count} 个文件")
```

### 逐行处理大文件

```python
# 对于超大文件（GB 级），不要一次性 read_text()
# 改用逐行读取，节省内存
def process_large_file(file_path: str) -> None:
    with open(file_path, encoding="utf-8") as f:
        for line_no, line in enumerate(f, start=1):
            # 逐行处理，内存只占用一行
            if "ERROR" in line:
                print(f"L{line_no}: {line.rstrip()}")
```

## 常见错误

### 1. UnicodeDecodeError

```python
# ❌ 错误：未指定编码，Windows 上中文文件报错
content = open("report.txt").read()

# ✅ 正确：明确指定 UTF-8
content = open("report.txt", encoding="utf-8").read()
# 或用 pathlib
content = Path("report.txt").read_text(encoding="utf-8")
```

### 2. 写入后文件内容丢失

```python
# ❌ 错误："w" 模式会覆盖整个文件
with open("log.txt", "w") as f:
    f.write("新内容")  # 原有内容被清空

# ✅ 追加用 "a" 模式
with open("log.txt", "a", encoding="utf-8") as f:
    f.write("追加内容\n")
```

### 3. 替换后忘记写回文件

```python
# ❌ 错误：read_text + replace 不会修改文件
content = Path("file.txt").read_text()
content.replace("old", "new")  # 字符串是不可变的，原地替换没有效果

# ✅ 正确：replace 返回新字符串，需要写回
content = Path("file.txt").read_text(encoding="utf-8")
new_content = content.replace("old", "new")
Path("file.txt").write_text(new_content, encoding="utf-8")
```
