## 第二部分 - 实现步骤

### 2.1 Pillow 基础

#### 概念说明

`PIL.Image` 是 Pillow 的核心类，代表一张图片。

```
┌─────────────────────────────────────────────────────────────┐
│                  Image 对象常用属性和方法                     │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  img.size      →  (width, height) 元组，如 (1920, 1080)     │
│  img.format    →  "JPEG" / "PNG" / "WEBP" ...               │
│  img.mode      →  "RGB" / "RGBA" / "L"（灰度）...            │
│                                                             │
│  img.save(path)         保存到文件                           │
│  img.resize((w, h))     缩放到指定尺寸                       │
│  img.convert("RGB")     转换色彩模式                         │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

#### 示例代码

```python
# image_basics.py
from PIL import Image


def image_info(file_path: str) -> None:
    """打印图片基本信息"""
    with Image.open(file_path) as img:
        print(f"文件：{file_path}")
        print(f"尺寸：{img.width} × {img.height}")
        print(f"格式：{img.format}")
        print(f"模式：{img.mode}")
        print(f"大小：{img.width * img.height:,} 像素")


image_info("/tmp/photo.jpg")
# 文件：/tmp/photo.jpg
# 尺寸：1920 × 1080
# 格式：JPEG
# 模式：RGB
```

### 2.2 格式转换

#### 概念说明

`Image.save(path, format)` 保存时自动根据扩展名或 `format` 参数决定输出格式。

**注意：PNG 支持透明（RGBA 模式），JPG 不支持透明。PNG → JPG 时需要先将 RGBA 转为 RGB。**

#### 示例代码

```python
# convert_format.py
from pathlib import Path
from PIL import Image


def convert_image(
    input_path: str,
    output_path: str,
    quality: int = 85,
) -> None:
    """转换图片格式

    Args:
        input_path: 原始图片路径
        output_path: 输出路径（扩展名决定格式）
        quality: JPEG/WebP 质量，1-100（PNG 不使用）
    """
    with Image.open(input_path) as img:
        output_ext = Path(output_path).suffix.lower()

        # PNG → JPG 时，透明通道（Alpha）需要合并到白色背景
        if output_ext in (".jpg", ".jpeg") and img.mode == "RGBA":
            background = Image.new("RGB", img.size, (255, 255, 255))
            background.paste(img, mask=img.split()[3])  # 用 Alpha 通道作为遮罩
            img = background
        elif img.mode not in ("RGB", "L"):
            img = img.convert("RGB")

        save_kwargs: dict = {}
        if output_ext in (".jpg", ".jpeg", ".webp"):
            save_kwargs["quality"] = quality
        if output_ext == ".webp":
            save_kwargs["method"] = 6  # 压缩质量最高（速度最慢）

        img.save(output_path, **save_kwargs)


# PNG 转 JPG
convert_image("/tmp/logo.png", "/tmp/logo.jpg", quality=90)

# JPG 转 WebP（体积通常比 JPG 小 30%）
convert_image("/tmp/photo.jpg", "/tmp/photo.webp", quality=80)
```

**关键代码说明：**

| 代码 | 含义 | 为什么这样写 |
|------|------|-------------|
| `img.mode == "RGBA"` | 检查图片是否含透明通道 | JPG 不支持透明，直接保存会报错 |
| `Image.new("RGB", img.size, (255, 255, 255))` | 创建白色背景图 | 将透明区域填充为白色 |
| `background.paste(img, mask=img.split()[3])` | 将原图贴到白色背景，用 Alpha 通道控制透明度 | `split()[3]` 提取 RGBA 中的第 4 通道（Alpha） |
| `save_kwargs["method"] = 6` | WebP 最高压缩质量 | method 0-6，越大压缩越好但越慢 |

### 2.3 批量格式转换

#### 示例代码

```python
# batch_convert.py
from pathlib import Path
from PIL import Image


def batch_convert(
    directory: str,
    source_ext: str,
    target_ext: str,
    quality: int = 85,
    delete_original: bool = False,
) -> list[str]:
    """批量转换目录下指定格式的图片

    Returns:
        转换记录列表
    """
    dir_path = Path(directory)
    records = []

    for file in dir_path.glob(f"*{source_ext}"):
        if not file.is_file():
            continue

        output_path = file.with_suffix(target_ext)
        try:
            convert_image(str(file), str(output_path), quality=quality)
            records.append(f"{file.name} → {output_path.name}")

            if delete_original:
                file.unlink()
        except Exception as e:
            records.append(f"失败 {file.name}: {e}")

    return records


# 将目录下所有 PNG 转为 WebP
records = batch_convert(
    "/tmp/images",
    source_ext=".png",
    target_ext=".webp",
    quality=80,
)
for r in records:
    print(r)
```

### 2.4 批量压缩图片

#### 概念说明

JPEG 的 `quality` 参数（1-100）控制压缩质量：85 是视觉无损的常用值，60 会有明显压缩痕迹。

#### 示例代码

```python
# compress_images.py
from pathlib import Path
from PIL import Image


def compress_image(
    input_path: str,
    output_path: str,
    max_size_kb: int = 500,
    quality_start: int = 85,
) -> int:
    """压缩图片到指定大小以内，返回最终 quality 值"""
    quality = quality_start

    with Image.open(input_path) as img:
        if img.mode != "RGB":
            img = img.convert("RGB")

        while quality > 10:
            img.save(output_path, "JPEG", quality=quality)
            size_kb = Path(output_path).stat().st_size // 1024
            if size_kb <= max_size_kb:
                break
            quality -= 5

    return quality


def compress_directory(
    directory: str,
    max_size_kb: int = 500,
) -> list[str]:
    """批量压缩目录下所有 JPG/JPEG 文件"""
    records = []
    for file in Path(directory).glob("*.jpg"):
        original_kb = file.stat().st_size // 1024
        final_quality = compress_image(str(file), str(file), max_size_kb)
        final_kb = file.stat().st_size // 1024
        records.append(
            f"{file.name}: {original_kb}KB → {final_kb}KB (quality={final_quality})"
        )
    return records
```

### 2.5 批量缩放图片

#### 示例代码

```python
# resize_images.py
from pathlib import Path
from PIL import Image


def resize_image(
    input_path: str,
    output_path: str,
    max_width: int,
    max_height: int,
) -> tuple[int, int]:
    """等比例缩放图片，确保不超过最大宽高，返回最终尺寸"""
    with Image.open(input_path) as img:
        # 等比例计算缩放后的尺寸
        img.thumbnail((max_width, max_height), Image.LANCZOS)
        img.save(output_path)
        return img.size


def resize_directory(
    directory: str,
    max_width: int = 1920,
    max_height: int = 1080,
    extensions: list[str] | None = None,
) -> list[str]:
    """批量缩放目录下图片，保持等比"""
    if extensions is None:
        extensions = [".jpg", ".jpeg", ".png", ".webp"]

    records = []
    for file in Path(directory).iterdir():
        if not file.is_file() or file.suffix.lower() not in extensions:
            continue
        try:
            with Image.open(str(file)) as img:
                original_size = img.size
            final_size = resize_image(str(file), str(file), max_width, max_height)
            records.append(
                f"{file.name}: {original_size[0]}×{original_size[1]} → "
                f"{final_size[0]}×{final_size[1]}"
            )
        except Exception as e:
            records.append(f"失败 {file.name}: {e}")
    return records
```

**关键代码说明：**

| 代码 | 含义 | 为什么这样写 |
|------|------|-------------|
| `img.thumbnail((max_width, max_height), Image.LANCZOS)` | 等比例缩放，不超过指定尺寸 | `thumbnail` 自动保持宽高比，`LANCZOS` 是高质量缩放算法 |
| `Image.LANCZOS` | 高质量缩放滤镜 | 比默认的 `NEAREST` 效果更好，适合照片缩放 |
| `img.thumbnail` vs `img.resize` | `thumbnail` 等比例缩小，`resize` 强制拉伸 | 批量缩放用 `thumbnail` 不会变形 |

## 第三部分 - 格式选择指南

| 格式 | 优点 | 缺点 | 适用场景 |
|------|------|------|---------|
| JPEG | 压缩率高，体积小 | 有损压缩，不支持透明 | 照片、写实图像 |
| PNG | 无损压缩，支持透明 | 体积比 JPG 大 | Logo、截图、含透明背景的图 |
| WebP | 比 JPG/PNG 体积小 30-50% | 旧版浏览器不支持 | Web 图片优化 |

## 常见错误

### 1. PNG 转 JPG 报错

```python
# ❌ 错误：RGBA 图片直接保存为 JPG 会报 OSError
img = Image.open("logo.png")  # mode = "RGBA"
img.save("logo.jpg")  # OSError: cannot write mode RGBA as JPEG

# ✅ 正确：先转换模式
img = img.convert("RGB")
img.save("logo.jpg")
```

### 2. resize 导致图片变形

```python
# ❌ 错误：强制拉伸到指定尺寸，图片会变形
img.resize((800, 600))  # 如果原图不是 4:3 比例，会变形

# ✅ 正确：用 thumbnail 保持等比
img.thumbnail((800, 600), Image.LANCZOS)
```

### 3. 忘记关闭图片对象

```python
# ❌ 错误：大量图片处理时可能内存溢出
img = Image.open("photo.jpg")
img.save("output.jpg")
# 没有关闭，img 一直占用内存

# ✅ 正确：用 with 语句自动关闭
with Image.open("photo.jpg") as img:
    img.save("output.jpg")
```
